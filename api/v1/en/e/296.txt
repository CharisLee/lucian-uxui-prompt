Element: Scrubbable contact index
ID: 296 · Category: Navigation · Tags: index, scrubber, contacts, list, magnify
What it does: Scrub up and down the letter index at the right of a contact list and a large letter bubble follows to the left of your fingertip as the list jumps straight to that letter group. Index letters near the fingertip bow out to the left and grow so you can see exactly where you are, and when you let go the bubble fades while the group header stays pinned at the top.
How to use it: Drag up and down the index bar on the right. Hovering works too
Footprint: fixed size, tall

<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var ls = root.querySelector('.u296-ls'), ix = root.querySelector('.u296-ix'), bub = root.querySelector('.u296-bub'), pin = root.querySelector('.u296-pin');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var IX = ['A', 'B', 'C', 'D', 'E', 'G', 'H', 'J', 'K', 'L', 'M', 'N', 'P', 'S'];
  var PEOPLE = [['Aaron Kim', 'Ada Park', 'Alex Han', 'Amy Lee'], ['Ben Cho', 'Bora Yun'], ['Chloe Do', 'Chris Seo'], ['Dami Ryu', 'Daniel Ra'], ['Eli Moon', 'Emma Min', 'Evan Ma'], ['Grace Park', 'Gwen Bae', 'Gyu Baek'], ['Hana Seo', 'Harin Song', 'Hugo Shin'], ['Jay Oh', 'Jia Yun', 'Jin Lee', 'June Lim'], ['Kai Jang', 'Kate Jung', 'Kevin Jo'], ['Lena Cha', 'Leo Choi'], ['Mia Koo', 'Milo Kang'], ['Nate Tae', 'Nora Tak'], ['Paul Pyo', 'Penny Pyeon'], ['Sam Han', 'Sofia Hong', 'Sora Heo']];
  var COL = ['#FF8787', '#FFA94D', '#FFD43B', '#69DB7C', '#38D9A9', '#4DABF7', '#748FFC', '#B197FC', '#F783AC'];
  var TOPS = [], y = 0, ci = 0;
  PEOPLE.forEach(function (g, i) {
    var h = document.createElement('div'); h.className = 'u296-gh'; h.textContent = IX[i]; ls.appendChild(h); TOPS.push(y); y += 26;
    g.forEach(function (n) { var p = document.createElement('div'); p.className = 'u296-p'; p.innerHTML = '<i style="background:' + COL[ci++ % COL.length] + '">' + n.charAt(0) + '</i>' + n; ls.appendChild(p); y += 46; });
  });
  var MAXY = Math.max(0, y - 424), K = IX.map(function (l, i) {
    var b = document.createElement('button'); b.type = 'button'; b.className = 'u296-k'; b.textContent = l; b.setAttribute('aria-label', l + ' group, jump');
    b.addEventListener('click', function () { go(i, false); });
    ix.appendChild(b); return b;
  });
  var cur = -1;
  function go(i, fast) {
    i = Math.max(0, Math.min(IX.length - 1, i));
    if (i === cur) return;
    cur = i;
    ls.classList.toggle('is-fast', !!fast);
    ls.style.transform = 'translateY(' + (-Math.min(MAXY, TOPS[i])) + 'px)';
    K.forEach(function (b, j) { b.classList.toggle('is-on', j === i); });
    pin.textContent = IX[i];
    bub.querySelector('b').textContent = IX[i];
  }
  function scrub(e, on) {
    var r = ix.getBoundingClientRect(), k = ix.offsetHeight / (r.height || 1), yy = (e.clientY - r.top) * k, h = ix.offsetHeight;
    var f = Math.max(0, Math.min(0.9999, yy / h)), i = Math.floor(f * IX.length);
    go(i, true);
    var rb = root.getBoundingClientRect(), kk = root.offsetWidth / (rb.width || 1);
    bub.style.top = ((e.clientY - rb.top) * kk).toFixed(1) + 'px'; bub.classList.toggle('is-on', on);
    K.forEach(function (b, j) {
      var cy = (j + 0.5) / IX.length * h, d = Math.abs(cy - yy), w = on ? Math.max(0, 1 - d / 70) : 0, s = Math.sin(w * Math.PI / 2);
      b.style.setProperty('--dx', (-s * 26).toFixed(1) + 'px'); b.style.setProperty('--sc', (1 + s * 0.6).toFixed(2));
    });
  }
  function rest() { bub.classList.remove('is-on'); K.forEach(function (b) { b.style.setProperty('--dx', '0px'); b.style.setProperty('--sc', 1); }); }
  var down = false;
  ix.addEventListener('pointerdown', function (e) { down = true; try { ix.setPointerCapture(e.pointerId); } catch (_) {} scrub(e, true); });
  ix.addEventListener('pointermove', function (e) { scrub(e, true); });
  function up() { down = false; rest(); }
  ix.addEventListener('pointerup', up); ix.addEventListener('pointercancel', up);
  ix.addEventListener('pointerleave', function () { if (!down) rest(); });
  if (rm) K.forEach(function (b) { b.style.transition = 'none'; });
  go(0, false);
})();
</script>
