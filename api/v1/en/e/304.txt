<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var q = root.querySelector('.u304-q'), rs = root.querySelector('.u304-rs'), zp = root.querySelector('.u304-zp'), at = root.querySelector('.u304-at'), ad = root.querySelector('.u304-ad'), dt = root.querySelector('.u304-dt'), go = root.querySelector('.u304-go');
  var uid = 'u304' + Math.random().toString(36).slice(2, 7);
  dt.id = uid; root.querySelector('.u304-pk label').htmlFor = uid;
  var A = [['94103', '1355 Market St, San Francisco, CA', 'Market Square'], ['94102', '1 Market St, San Francisco, CA', 'Spear Tower'], ['94105', '101 Market St, San Francisco, CA', 'One Market Plaza'], ['10001', '20 W 34th St, New York, NY', 'Empire State Building']];
  function mark(el, text, k) {
    var i = k ? text.toLowerCase().indexOf(k.toLowerCase()) : -1;
    if (i < 0) { el.appendChild(document.createTextNode(text)); return; }
    el.appendChild(document.createTextNode(text.slice(0, i))); var b = document.createElement('b'); b.textContent = text.slice(i, i + k.length); el.appendChild(b); el.appendChild(document.createTextNode(text.slice(i + k.length)));
  }
  function search() {
    var k = q.value.trim(); rs.innerHTML = '';
    if (k.length < 2) { root.classList.remove('is-list'); return; }
    var hit = A.filter(function (a) { return (a[1] + a[2]).toLowerCase().indexOf(k.toLowerCase()) >= 0; });
    var s = document.createElement('small'); s.textContent = hit.length ? hit.length + ' results' : ''; rs.appendChild(s);
    if (!hit.length) { var n = document.createElement('div'); n.className = 'u304-r'; n.innerHTML = '<span>No results</span>'; rs.appendChild(n); }
    hit.forEach(function (a) {

      var em = document.createElement('em'); em.textContent = a[0]; b.appendChild(em); mark(b, a[1], k);
      var sp = document.createElement('span'); mark(sp, a[2], k); b.appendChild(sp);
      b.addEventListener('click', function () { pick(a); });
      rs.appendChild(b);
    });
    root.classList.add('is-list');
  }
  function pick(a) {
    root.classList.remove('is-list'); zp.textContent = a[0]; at.textContent = a[1];

    sync();
  }
  function sync() { go.disabled = !(root.classList.contains('is-pick') && dt.value.trim()); }

  dt.addEventListener('input', sync);
  root.addEventListener('keydown', function (e) { if (e.key === 'Escape') root.classList.remove('is-list'); });
})();
